// ---------- Markdown-рендер ----------
const { MarkdownRenderer, Component } = require("obsidian");
const mdComponent = new Component();
mdComponent.load();

// Рендерит markdown в элемент и убирает блочные отступы у <p>
async function renderPiece(el, md) {
    if (MarkdownRenderer) {
        await MarkdownRenderer.render(app, md, el, "", mdComponent);
        el.querySelectorAll("p").forEach(p => {
            p.style.margin = "0";
            p.style.display = "inline";
        });
    } else {
        el.textContent = md; // fallback
    }
}

// ---------- Библиотека ----------
function resolveFile(path) {
    const currentPath = app.workspace.getActiveFile()?.path || "";
    return app.metadataCache.getFirstLinkpathDest(path, currentPath);
}
function parseRow(line) {
    const cells = [];
    let cur = "";
    for (let i = 0; i < line.length; i++) {
        const ch = line[i];
        if (ch === "\\" && line[i + 1] === "|") { cur += "|"; i++; }
        else if (ch === "|") { cells.push(cur); cur = ""; }
        else cur += ch;
    }
    cells.push(cur);
    return cells.slice(1, -1).map(c => c.trim());
}
async function getTableRows(file, blockId) {
    const content = await app.vault.read(file);
    const lines = content.split("\n");
    const idIndex = lines.findIndex(l => l.trim() === blockId);
    if (idIndex === -1) return null;
    const tableLines = [];
    for (let i = idIndex - 1; i >= 0; i--) {
        const line = lines[i].trim();
        if (line.startsWith("|")) tableLines.unshift(line);
        else if (line === "" && tableLines.length) break;
        else if (line !== "") break;
    }
    if (tableLines.length < 3) return [];
    return tableLines.slice(2).map(parseRow);
}
const randomItem = arr => arr[Math.floor(Math.random() * arr.length)];

async function randomFromTable(file, blockId, columns) {
    const rows = await getTableRows(file, blockId);
    if (!rows || rows.length === 0) return null;
    const row = randomItem(rows);
    return columns.map(c => row[c] ?? null);
}

function makeRollerButton(container, options) {
    const {
        file,
        blockIds,
        blockPrefix = "",
        buttonText = "🎲 Roll",
        column = [1],
        separator = "·",
        errorColor = "var(--text-error)",
        containerStyle = "margin-top: 8px; display: flex; flex-wrap: wrap; gap: 8px; font-size: 1.1em; align-items: center;",
    } = options;

    const columns = Array.isArray(column) ? column : [column];

    const ids = (Array.isArray(blockIds) ? blockIds : [blockIds]).map(id =>
        id.startsWith("^") ? id : blockPrefix + id
    );

    const btn = container.createEl("button", { text: buttonText });
    const out = container.createEl("div", { attr: { style: containerStyle } });

    async function roll() {
        out.empty();
        const pieces = []; // { text, isError }

        for (let i = 0; i < ids.length; i++) {
            const values = await randomFromTable(file, ids[i], columns);

            if (!values || values.every(v => v === null || v === "")) {
                pieces.push({ text: `[${ids[i]} missing/empty]`, isError: true });
            } else {
                for (const v of values) {
                    if (v !== null && v !== "") pieces.push({ text: v, isError: false });
                }
            }
        }

        for (let idx = 0; idx < pieces.length; idx++) {
            const p = pieces[idx];
            if (p.isError) {
                out.createEl("span", {
                    text: p.text,
                    style: `color: ${errorColor}; font-size: 0.9em;`,
                });
            } else {
                const span = out.createEl("span", {
                    attr: { style: "display: inline-flex; align-items: center;" },
                });
                await renderPiece(span, p.text);
            }
            if (idx < pieces.length - 1) {
                out.createEl("span", { text: separator, style: "color: var(--text-muted);" });
            }
        }
    }

    btn.addEventListener("click", roll);
    return { button: btn, output: out, roll };
}
